function calculate_money_fix(_money_borrow_fix, _month_borrow_fix, _interest_rate_fix)
{
    var money_borrow_fix = parseInt(_money_borrow_fix);
    var month_borrow_fix = parseInt(_month_borrow_fix);
    var interest_rate_fix = parseFloat(_interest_rate_fix);

    var money_interest_month_fix = 0;
    var array_average_month_fix = [];
    var money_pay_month_fix = 0;

    var money_origin_month_fix = money_borrow_fix / month_borrow_fix;
    var money_remain_fix = money_borrow_fix;

    var x = 0;
    while(x < month_borrow_fix)
    {
        // tien lai 1 thang
        money_interest_month_fix = money_borrow_fix * interest_rate_fix / month_borrow_fix;

        // tien phai tra moi thang
        money_pay_month_fix = money_origin_month_fix + money_interest_month_fix;

        // so goc con lai
        money_remain_fix = money_remain_fix - money_origin_month_fix;
        
        array_average_month_fix.push({
        "money_interest_month"  : Math.round(money_interest_month_fix),
        "money_origin_month"    : Math.round(money_origin_month_fix),
        "money_pay_month"       : Math.round(money_pay_month_fix),
        "money_borrow"          : Math.round(money_remain_fix)
        });
        x++;
    }

    generate_html(array_average_month_fix);
}
function calculate_money_borrow(_money_borrow, _month_borrow, _interest_rate)
{
    var money_borrow = parseInt(_money_borrow);
    var month_borrow = parseInt(_month_borrow);
    var interest_rate = parseFloat(_interest_rate);

    // tien lai 1 thang
    var money_interest_month = 0;
    var array_average_month = [];
    // tien phai tra moi thang
    var money_pay_month = 0;
    var is_first = true;

    var money_origin_month = money_borrow / month_borrow;

    var i = 0;
    while(i < month_borrow)
    {
        if( is_first )
        {
            money_interest_month = (money_borrow *  interest_rate) / month_borrow;
            is_first = false;       
        }
        else {
            money_interest_month = ( money_borrow * interest_rate ) / month_borrow;
        }
        
        money_borrow = money_borrow - money_origin_month;
        money_pay_month  = money_origin_month + money_interest_month;

        array_average_month.push({
            "money_interest_month"  : Math.round(money_interest_month),
            "money_origin_month"    : Math.round(money_origin_month),
            "money_pay_month"       : Math.round(money_pay_month),
            "money_borrow"          : Math.round(money_borrow)
        })
        i++;
    }
    generate_html(array_average_month)
}
