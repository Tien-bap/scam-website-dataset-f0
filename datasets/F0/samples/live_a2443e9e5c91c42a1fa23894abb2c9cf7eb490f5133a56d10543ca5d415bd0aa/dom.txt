<script>
    document.addEventListener("DOMContentLoaded", function() {
        var imgSrc = "";
        var popup1Link = "";

        if (imgSrc && popup1Link) {
            var popup1 = document.createElement("div");
            popup1.id = "popup1";
            popup1.className = "popup1";
            popup1.style.display = "none";

            var img = document.createElement("img");
            img.src = imgSrc;
            img.alt = "popup1 Image";

            img.addEventListener("click", function() {
                window.location.href = popup1Link; 
            });

            popup1.appendChild(img);

            var closeBtn = document.createElement("button");
            closeBtn.id = "closeBtn";
            closeBtn.textContent = "Đóng";

            closeBtn.addEventListener("click", function() {
                popup1.style.display = "none";
                sessionStorage.setItem("popup1ClosedTime", Date.now());
            });

            popup1.appendChild(closeBtn);
            document.body.appendChild(popup1);

            var closedTime = sessionStorage.getItem("popup1ClosedTime");
            if (!closedTime || (Date.now() - closedTime > 10000)) {
                popup1.style.display = "block";
            }

            setTimeout(function() {
                popup1.style.display = "none";
                sessionStorage.setItem("popup1ClosedTime", Date.now());
            }, 1 * 60 * 1000); // 5 phút tính bằng mili giây
        }
    });
</script>
