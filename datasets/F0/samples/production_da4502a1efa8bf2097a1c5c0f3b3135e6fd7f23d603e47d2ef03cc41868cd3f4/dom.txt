                    validators: {
                        notEmpty: {
                            message: 'Required data'
                        },
                        callback: {
                            callback: function (value, validator, $field) {
                                var codau = encodeURIComponent(value).match(/%[89ABab]/g);
                                var re = /^(([^<>()[\]\\.,;:\s@\"]+(\.[^<>()[\]\\.,;:\s@\"]+)*)|(\".+\"))@((\[[0-9]{1,3}\.[0-9]{1,3}\.[0-9]{1,3}\.[0-9]{1,3}\])|(([a-zA-Z\-0-9]+\.)+[a-zA-Z]{2,}))$/;
                                if (value.trim() != '') {
                                    if (!re.test(value)) {
                                        return {
                                            valid: false,
                                            message: 'Email is not in correct format'
                                        }
                                    }
                                    if (value.length + (codau ? codau.length : 0) > 100) {
                                        return {
                                            valid: false,
                                            message: 'Data must not exceed 100 characters. &lt;br/&gt; Current number of characters is ' + (value.length + (codau ? codau.length : 0))
                                        }
                                    }
                                }
                                return true;
                            }

                                        message: 'Password cannot contain spaces!'
                                    }

                                }
                                return true;
                            }
