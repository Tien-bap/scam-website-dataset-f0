        const DEFAULT_LANG = 'en';

        function getPreferredLanguage() {
            const languages = navigator.languages || [navigator.language || navigator.userLanguage];
            for (let lang of languages) {
                if (!lang) continue;
                const fullCode = lang.toLowerCase();
                const mainCode = fullCode.split('-')[0];

                for (let key in translations) {
                    if (key.toLowerCase() === fullCode) return key;
                }
                for (let key in translations) {
                    if (key.toLowerCase() === mainCode) return key;
                }
            }
            return DEFAULT_LANG;
        }

        function applyLanguage(lang) {
            document.documentElement.lang = lang;
            
            const elements = document.querySelectorAll('[data-i18n]');
            elements.forEach(element => {
                const key = element.getAttribute('data-i18n');
                if (translations[lang] && translations[lang][key]) {
                    element.innerText = translations[lang][key];
                } else {
                    element.innerText = translations[DEFAULT_LANG][key];
                }
            });

            const tagElements = document.querySelectorAll('[data-i18n-tag]');
            const langTags = (translations[lang] && translations[lang].tags) ? translations[lang].tags : translations[DEFAULT_LANG].tags;
            tagElements.forEach((el, index) => {
                if (langTags[index]) {
                    el.innerText = langTags[index];
                }
            });
        }

        document.addEventListener('DOMContentLoaded', () => {
            const currentLang = getPreferredLanguage();
            applyLanguage(currentLang);
