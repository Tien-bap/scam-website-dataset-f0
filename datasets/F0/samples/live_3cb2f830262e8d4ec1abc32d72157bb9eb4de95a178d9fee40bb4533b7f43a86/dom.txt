  function hdshowmore(e) {
    $(".item_contentshow_more").removeClass("activehd");
    $(".box_contentshow_more .header_content_more").hide();
    var data_s = $(e).attr("data-s");
    if (data_s == 0) {
      $(".item_contentshow_more").attr("data-s", 0);
      $(e).addClass("activehd");
      $(e).attr("data-s", 1);
      $(e).parents(".box_contentshow_more").find(".header_content_more").show();
    } else {
      $(e).removeClass("activehd");
      $(e).attr("data-s", 0);
      $(e).find(".header_content_more").hide();
    }
  }
