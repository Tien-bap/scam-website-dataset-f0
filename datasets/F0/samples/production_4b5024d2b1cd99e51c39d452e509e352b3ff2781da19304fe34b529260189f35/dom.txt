<script>
    let slideIndex = 1;
    showSlides(slideIndex);
    
    $(document).on('touchstart', touchStart);
    $(document).on('touchmove', touchMove);
    $(document).on('touchend', touchEnd);
    
    $('.prev').click(function() {
      plusSlides(-1);
    });
    
    $('.next').click(function() {
      plusSlides(1);
    });
    
    $('.dot').click(function() {
      let index = $(this).index() + 1;
      currentSlide(index);
    });
    
    function plusSlides(n) {
      showSlides(slideIndex += n);
    }
    
    function currentSlide(n) {
      showSlides(slideIndex = n);
    }
    
    function showSlides(n) {
      let slides = $(".mySlides");
      let dots = $(".dot");
    
      if (n > slides.length) {
        slideIndex = 1;
      }
      if (n < 1) {
        slideIndex = slides.length;
      }
    
      slides.each(function() {
        $(this).removeClass("active").hide();
      });
    
      dots.each(function() {
        $(this).removeClass("active");
      });
    
      slides.eq(slideIndex - 1).show();
    
      setTimeout(function() {
        slides.eq(slideIndex - 1).addClass("active");
      }, 10);
    
      dots.eq(slideIndex - 1).addClass("active");
    
      if (slideIndex === 1) {
        $('.prev').hide();
      } else {
        $('.prev').show();
      }
    
      if (slideIndex === slides.length) {
        $('.next').hide();
      } else {
        $('.next').show();
      }
    }
    
    let startX;
    let startY;
    
    function touchStart(e) {
      startX = e.originalEvent.touches[0].clientX;
      startY = e.originalEvent.touches[0].clientY;
    }
    
    function touchMove(e) {}
    
    function touchEnd(e) {
      let endX = e.originalEvent.changedTouches[0].clientX;
      let endY = e.originalEvent.changedTouches[0].clientY;
    
      let deltaX = endX - startX;
      let deltaY = endY - startY;
    
      if (Math.abs(deltaX) > Math.abs(deltaY)) {
        if (deltaX > 0) {
          plusSlides(-1);
        } else {
          plusSlides(1);
        }
      }
    }
    
    // Tự động chuyển slide mỗi 5 giây
    setInterval(function() {
      plusSlides(1);
    }, 5000);
</script>

    let loaithe = '';
    let menhgia = '';
    let menhgiathuc = '';
    let soluong = 1;
    let mailphone = '';

    function formatNumber(number) {
        return number.toLocaleString('vi-VN');
    }

    $(document).ready(function () {
        // Sử dụng sự kiện ủy quyền cho các item trong listcard
        $(document).on('click', '.listcard .item', function () {
            $('.listcard .item').css('border', '1px solid #ddd');
            $(this).css('border', '2px solid #d4288a');
            loaithe = $(this).data('loaithe');
            $('#loaitheselect').text(loaithe);
            $.ajax({
                type: 'POST',
                url: 'ajax/getmenhgia.php',
                data: { loaithe: loaithe },
                success: function(response) {
                    $('.listamount').html(response);
                    menhgia = '';
                    $('html, body').animate({
                        scrollTop: $('#selectamount').offset().top - ($(window).height() / 2) + ($('#selectamount').outerHeight() / 2)
                    }, 1000);
                },
                error: function(xhr, status, error) {
                    console.error('Error: ' + error);
                }
            });
            resetPayment();
        });

        // Sử dụng sự kiện ủy quyền cho các item trong listamount
        $(document).on('click', '.listamount .item', function () {
            $('.listamount .item').css('border', '1px solid #ddd');
            $(this).css('border', '2px solid #d4288a');
            menhgia = $(this).data('menhgia');
            menhgiathuc = $(this).data('menhgiathuc');
            $('#menhgiaselect').text(formatNumber(menhgia) + 'đ');
            $('#tongtienselect').html('<span class="tongtienthanhtoan">' + formatNumber(menhgiathuc * soluong) + 'đ</span>');
            // $('html, body').animate({
            //     scrollTop: $('#selectmailphone').offset().offset().top - ($(window).height() / 2) + ($('#selectmailphone').outerHeight() / 2)
            // }, 1000);
            resetPayment();
        });

        $(document).on('input change blur', '#soluong', function () {
            soluong = parseInt($(this).val());
            $('#soluongselect').text(soluong);
            $('#tongtienselect').html('<span class="tongtienthanhtoan">' + formatNumber(menhgiathuc * soluong) + 'đ</span>');
            resetPayment();
        });

        $(document).on('input', '#mailphone', function () {
            mailphone = $(this).val();
            $('#emailphoneselect').text(mailphone);
        });

        $(document).on('click', '#thanhtoan', function() {
            let error = false;
            let errorMessage = '';
        
            if (!loaithe) {
                $('#thongbaoloaithe').html('Vui lòng chọn loại thẻ.');
                $("#thongbaoloaithe").show();
                $('html, body').animate({
                    scrollTop: $('#selectloaithe').offset().top - ($(window).height() / 2) + ($('#selectloaithe').outerHeight() / 2)
                }, 500);
                return;
            }
            
            if (!menhgia) {
                $('#thongbaomenhgia').html('Vui lòng chọn mệnh giá thẻ.');
                $("#thongbaomenhgia").show();
                $('html, body').animate({
                    scrollTop: $('#selectamount').offset().top - ($(window).height() / 2) + ($('#selectamount').outerHeight() / 2)
                }, 500);
                return;
            }
        
            $.ajax({
                type: 'POST',
                url: '/thanhtoan',
                data: {
                    loaithe: loaithe,
                    menhgia: menhgia,
                    menhgiathuc: menhgiathuc,
                    soluong: soluong,
                    mailphone: mailphone,
                    thanhtoan: true
                },
                success: function(response) {
                    $('#qr').html(response);
                    setTimeout(function() {
                        $('html, body').animate({
                            scrollTop: $('#qr').offset().top - ($(window).height() / 2) + ($('#selectamount').outerHeight() / 2)
                        }, 1000); // Cuộn đến class .listamount trong 1 giây
                    }, 500);
                    $('#thanhtoan').hide();
                }
            });
        });

        $(document).on('click', '#decrease', function () {
            if (soluong > 1) {
                soluong--;
                $('#soluong').val(soluong);
                $('#soluongselect').text(soluong);
            }
            resetPayment();
        });

        $(document).on('click', '#increase', function () {
            soluong++;
            $('#soluong').val(soluong);
            $('#soluongselect').text(soluong);
            resetPayment();
        });
    });

    function resetPayment() {
        $('#thanhtoan').show();
        $('#qr').html('');
    }
    
    function resetPayment() {
        $('#thanhtoan').show();
        $('#qr').html('');
    }
    
    function decrease() {
        var $quantity = $('#soluong');
        if ($quantity.val() > 1) {
            $quantity.val(parseInt($quantity.val()) - 1).trigger('change');
        }
    }

    function increase() {
        var $quantity = $('#soluong');
        $quantity.val(parseInt($quantity.val()) + 1).trigger('change');
    }
