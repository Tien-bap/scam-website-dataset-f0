  (() => {
    const heroLink    = document.getElementById('afropoCoverHeroLink');
    const heroImg     = document.getElementById('afropoCoverHeroImg');
    const heroTitle   = document.getElementById('afropoCoverHeroTitle');
    const heroExcerpt = document.getElementById('afropoCoverHeroExcerpt');
    const list        = document.getElementById('afropoCoversList');

    if (!heroLink || !list) return;

    const items = Array.from(list.querySelectorAll('.home-covers__thumbitem'));

    function setHero(item) {
      const url     = item.dataset.url     || '#';
      const img     = item.dataset.img     || '';
      const title   = item.dataset.title   || '';
      const excerpt = item.dataset.excerpt || '';

      heroLink.href            = url;
      heroImg.src              = img;
      heroImg.alt              = title;
      heroTitle.textContent    = title;
      heroExcerpt.textContent  = excerpt;

      items.forEach(el => el.classList.remove('home-covers__thumbitem--active'));
      item.classList.add('home-covers__thumbitem--active');

      // Update thumb links so they go to the article (not intercept click)
      item.querySelector('a').href = url;
    }

    items.forEach(item => {
      const link = item.querySelector('.home-covers__thumblink');
      link.addEventListener('click', e => {
        // First click: activate hero. Second click (already active): follow link.
        if (!item.classList.contains('home-covers__thumbitem--active')) {
          e.preventDefault();
          setHero(item);
        }
        // if already active, let the link navigate normally
      });
    });

    // Initialise with first item
    if (items[0]) setHero(items[0]);
  })();
