    async function enterCloudMode(userId){
      const SC=window.ShoppeCloud; if(!SC) return;
      cloudMode=true;
      // migrate local cart to DB (best-effort, non-duplicating)
      try{
        const local=(cartItems||[]).filter(x=>Number.isInteger(x.i)&&x.i>=0);
        for(const it of local){ const p=products[it.i]; if(!p) continue; await SC.addLocalToCart({local_index:it.i,name:p[0],image:p[4],price:p[1],qty:it.qty}); }
        try{ localStorage.removeItem(CART_KEY); }catch(e){}
      }catch(e){ console.warn('[cart migrate]',e); }
      await refreshCloudCart();
      if(cartUnsub) cartUnsub();
      cartUnsub=SC.subscribeCart(userId, ()=>refreshCloudCart());
      if(ordersUnsub) ordersUnsub();
      ordersUnsub=SC.subscribeUserOrders(userId,(p)=>{
        if(p.event==='UPDATE'&&p.new?.status&&p.old?.status&&p.new.status!==p.old.status){

    /* Identifier detection: email > phone > username. Returns {type,valid,message}. */
    function detectIdentifier(raw){
      const v=(raw||"").trim();
      if(!v)return{type:"empty",valid:false,message:"Vui lòng nhập email, số điện thoại hoặc tên đăng nhập."};
      if(v.includes("@")){
        const ok=/^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/.test(v);
        return{type:"email",valid:ok,message:ok?"":"Email chưa đúng định dạng. Ví dụ: ten@example.com"};
      }
      if(/^[+\d][\d\s.-]*$/.test(v)){
        const digits=v.replace(/[\s.-]/g,"");
        const ok=/^(0\d{9}|\+84\d{9})$/.test(digits);
        return{type:"phone",valid:ok,message:ok?"":"Số điện thoại chưa đúng định dạng. Ví dụ: 0912345678"};
      }
      const ok=/^[a-zA-Z][a-zA-Z0-9._]{3,23}$/.test(v);
      return{type:"username",valid:ok,message:ok?"":"Tên đăng nhập cần 4–24 ký tự, bắt đầu bằng chữ, chỉ chứa chữ/số/._"};
    }
    function validateAuthForm(type){
      const account=document.querySelector("#authEmail")?.value||"";
      const pass=document.querySelector("#authPass")?.value||"";
      const confirmPass=document.querySelector("#authPassConfirm")?.value||"";
      const id=detectIdentifier(account);
      let message="",ok=false;
      if(!id.valid){message=id.message;}
      else if(!pass){message="Vui lòng nhập mật khẩu.";}
      else if(pass.length<6){message="Mật khẩu cần ít nhất 6 ký tự.";}
      else if(type==="Đăng ký"&&!confirmPass){message="Vui lòng nhập lại mật khẩu.";}
      else if(type==="Đăng ký"&&pass!==confirmPass){message="Hai lần nhập mật khẩu chưa giống nhau.";}
      else{ok=true;}
      return{ok,message,idType:id.type};
    }
