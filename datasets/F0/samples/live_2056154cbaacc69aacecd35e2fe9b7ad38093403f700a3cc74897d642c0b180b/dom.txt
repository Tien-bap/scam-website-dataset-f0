<script>
document.addEventListener('DOMContentLoaded', function () {
    const toggle = document.getElementById('gxMenuToggle');
    const overlay = document.getElementById('gxMobileMenu');
    const closeBtn = document.getElementById('gxMobileMenuClose');

    if (!toggle || !overlay) return;

    function openMenu() {
        overlay.classList.add('is-open');
        document.body.style.overflow = 'hidden';
    }

    function closeMenu() {
        overlay.classList.remove('is-open');
        document.body.style.overflow = '';
        document.querySelectorAll('#gxMobileMenu .gx-mobile-menu-item.is-open').forEach(function(item){
            item.classList.remove('is-open');
        });
    }

    toggle.addEventListener('click', function () {
        if (overlay.classList.contains('is-open')) {
            closeMenu();
        } else {
            openMenu();
        }
    });

    if (closeBtn) {
        closeBtn.addEventListener('click', closeMenu);
    }

    overlay.addEventListener('click', function (e) {
        if (e.target === overlay) closeMenu();
    });

    document.addEventListener('keydown', function (e) {
        if (e.key === 'Escape') closeMenu();
    });

    document.querySelectorAll('#gxMobileMenu .gx-mobile-dropdown-toggle').forEach(function(btn){
        btn.addEventListener('click', function(){
            const parent = btn.closest('.gx-mobile-menu-item');
            if (!parent) return;
            parent.classList.toggle('is-open');
        });
    });
});

<script>
$(document).ready(function () {

    function initMobileHomeSlider() {
        var $slider = $(".dc_home_slider");

        if (!$slider.length) return;

        $slider.css({
            display: "block",
            visibility: "visible",
            opacity: 1
        });

        if (typeof $.fn.owlCarousel === "function" && !$slider.hasClass("owl-loaded")) {
            $slider.owlCarousel({
                singleItem: true,
                items: 1,
                navigation: false,
                pagination: true,
                autoPlay: 8000,
                stopOnHover: true,
                slideSpeed: 500,
                paginationSpeed: 500,
                rewindSpeed: 500,
                autoHeight: true
            });
        }
    }

    initMobileHomeSlider();
    $(window).on("load resize", initMobileHomeSlider);

    function validate() {
        let ok = true;

        function err(id, msg) {
            $(id).addClass("is-invalid").attr("data-original-title", msg).tooltip("show");
            ok = false;
        }

        $(".form-control").removeClass("is-invalid").tooltip("dispose");

        if ($("#startPoint_tf").val().trim().length < 3) err("#startPoint_tf", "Vui lòng nhập điểm đi");
        if ($("#endPoint_tf").val().trim().length < 3) err("#endPoint_tf", "Vui lòng nhập điểm đến");

        let phone = $("#customer_phone_tf").val().replace(/[^0-9]/g, '');
        if (!/^[0-9]{10,11}$/.test(phone)) err("#customer_phone_tf", "Số điện thoại không hợp lệ");

        if ($("#customer_name_tf").val().trim().length < 2) {
            err("#customer_name_tf", "Vui lòng nhập họ tên hợp lệ");
        }

        return ok;
    }

    $("#bookNow_btn").click(function () {
        if (!validate()) return;

        $("#confirm_start_place").text($("#startPoint_tf").val());
        $("#confirm_end_place").text($("#endPoint_tf").val());
        $("#confirm_num_seats").text($("#numseats_sf option:selected").text());
        $("#confirm_pickup_time").text($("#pickuptime_tif").val());
        $("#confirm_phone").text($("#customer_phone_tf").val());
        $("#confirm_name").text($("#customer_name_tf").val());

        $("#confirm_round_trip_row").toggle($("#roundtrip_cf").is(":checked"));
        $("#confirm_vat_row").toggle($("#vat_cf").is(":checked"));

        $("#bookingModal").modal("show");
    });

    $("#confirm_btn").click(function () {
        $("#bookingModal").modal("hide");
        setTimeout(() => {
            $("#successModal").modal("show");
        }, 300);
    });

});
</script>

    const duration = isMobile ? 7000 : 10000;
    const offsetRatio = 0.45;
    const scale = isMobile ? 0.45 : 0.3;

    function getX(progress, bannerWidth, carWidth) {
        const totalDistance = bannerWidth + carWidth * 0.5;
        return bannerWidth + 10 - totalDistance * progress;
    }

    function animate(timestamp) {
        if (!isVisible) return;
        if (!start) start = timestamp;

        const bannerWidth = banner.offsetWidth;
        const car1Width = car1.offsetWidth;
        const car2Width = car2.offsetWidth;
        const elapsed = (timestamp - start) % duration;

        const progress1 = elapsed / duration;
        let progress2 = progress1 - offsetRatio;
        if (progress2 < 0) progress2 += 1;

        const x1 = getX(progress1, bannerWidth, car1Width);
        const x2 = getX(progress2, bannerWidth, car2Width);

        car1.style.transform = `translate3d(${x1}px, 0, 0) scale(${scale})`;
        car2.style.transform = `translate3d(${x2}px, 0, 0) scale(${scale})`;

        rafId = requestAnimationFrame(animate);
    }

    const observer = new IntersectionObserver((entries) => {
        entries.forEach(entry => {
            if (entry.isIntersecting) {
                if (!isVisible) {
                    isVisible = true;
                    rafId = requestAnimationFrame(animate);
                }
            } else {
                isVisible = false;
                if (rafId) cancelAnimationFrame(rafId);
            }
        });
    }, { threshold: 0.15 });

    observer.observe(banner);
});
        });
