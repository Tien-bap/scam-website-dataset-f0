function redirectToApp() {
    var userAgent = navigator.userAgent || navigator.vendor || window.opera;

    // 发送GET请求获取下载链接
    fetch('/api/get_download_url.php')
        .then(response => response.json())
        .then(data => {
            // 根据返回的数据更新下载链接并重定向
            if (/android/i.test(userAgent) || /HarmonyOS/i.test(userAgent)) {
                window.location.href = data.android_link;
            } else if (/iPad|iPhone|iPod/.test(userAgent) && !window.MSStream) {
                window.location.href = data.ios_link;
            } else {
                window.location.href = data.other_link;
            }
        })
        .catch(error => {
            console.error('Error fetching download URL:', error);
        });
}
