<script>

    $(document).on('click', '.ajax-upload-file .btn-submit', function(){
        let checkAgree = $('.agree').is(":checked"),
            checkNum = 0;
        if(checkAgree){
            var formData = new FormData();
            let TotalImages = cacheListFile.files.length; //Total Images
                for (let i = 0; i < TotalImages; i++) {
                    formData.append('listFile[]', cacheListFile.files[i]);
                }
            formData.append('_token', $(".ajax-upload-file input[name='_token']").val()); // CSRF token
            formData.append('fullname', $(".ajax-upload-file input[name='fullname']").val());
            formData.append('email', $(".ajax-upload-file input[name='email']").val());
            formData.append('message', $(".ajax-upload-file textarea[name='message']").val());
            formData.append('g-recaptcha-response', grecaptcha.getResponse());
            if(checkNum < 3){
                checkValidate(formData);
            }else{
                checkValidateCaptcha(formData)
            }
        }
        
    })

    function checkValidate(formData){
        $.ajax({
            
            url: "https://vn-ams.com/validate-subcribers",
            type: 'POST',
            processData: false,
            contentType: false,
            data: formData,
            success: function(data) {
                if (data.status == 200) {
                    loadFileUpLoad2(formData);
                } else {
                    $(".ajax-upload-file .print-error-msg").css('display', 'block');
                    $(".ajax-upload-file .print-success-msg").css('display', 'none');
                    $(".ajax-upload-file .print-error-msg span").html(data.error);
                }
            }
        });
    }

    function checkValidateCaptcha(formData){
        $.ajax({
            
            url: "https://vn-ams.com/validate-subcribers-captcha",
            type: 'POST',
            processData: false,
            contentType: false,
            data: formData,
            success: function(data) {
                if (data.status == 200) {
                    loadFileUpLoad2(formData);
                } else {
                    $(".ajax-upload-file .print-error-msg").css('display', 'block');
                    $(".ajax-upload-file .print-success-msg").css('display', 'none');
                    $(".ajax-upload-file .print-error-msg span").html(data.error);
                }
            }
        });
    }

    function loadFileUpLoad2(formData){

        $.ajax({
            xhr: function() {
                var xhr = new window.XMLHttpRequest();

                xhr.upload.addEventListener("progress", function(evt) {
                if (evt.lengthComputable) {
                    var percentComplete = evt.loaded / evt.total;
                    percentComplete = parseInt(percentComplete * 100);
                    $('.box-progressBar').removeClass('hidden').addClass('flex');
                    // $('.valueProgress').html(percentComplete + ' %');
                    $('.valueProgress').html('');
                    $('.valueProgress').append(percentComplete + ' %' + '<br> Uploading...');
                    $('.bg-progressBar').css('background', 'conic-gradient(#e91c2b ' + percentComplete * 3.6 + 'deg, #ccc 0deg)');
                }
                }, false);

                return xhr;
            },
            url: "https://vn-ams.com/subcribers",
            type: 'POST',
            processData: false,
            contentType: false,
            data: formData,
            success: function(data) {
                if (data.status == 200) {
                    $('.box-progressBar').addClass('hidden');
                    $('.box-thanks').removeClass('hidden').addClass('flex');
                    sendMail(formData);
                    setTimeout(function() {
                        location.reload();
                    }, 2000);
                }else{
                    $('.box-progressBar').addClass('hidden');
                    $('.valueProgress').html('');
                    $('.bg-progressBar').css('background', 'conic-gradient(#e91c2b 0deg, #ccc 0deg)');
                    $(".ajax-upload-file .print-error-msg").css('display', 'block');
                    $(".ajax-upload-file .print-success-msg").css('display', 'none');
                    $(".ajax-upload-file .print-error-msg span").html(data.error);
                }
            }
        });
    }

    function sendMail(formData){
        $.ajax({
            
            url: "https://vn-ams.com/send-mail",
            type: 'POST',
            processData: false,
            contentType: false,
            data: formData,
            success: function(data) {
                console.log('gửi emai thành công');
            }
        });
    }

    $(document).on('click', '.btn-upload-file.hasFile', function(){

    var cacheListFile = new DataTransfer();

    $(document).on('change', '#fileUpload', function(){
        $.each($(this)[0].files, function(key, item){
            cacheListFile.items.add(item);
        })
        loadData();
    })

    $(document).on('click', '.removeFile', function(){
        let sltRemove = $(this).attr('data-remove');
            cacheListFile.items.remove(sltRemove);
            loadData();
    });

    function loadData(){
        let arrFile = cacheListFile.files,
            textBtn = 'Get quote',
            html = '';
            $('.btn-upload-file').text('');
            $('.listFileUpload').text('');
        if(arrFile.length > 0){
            $('.btn-upload-file').addClass('hasFile');
            textBtn = textBtn + ' (' + arrFile.length + ')';
            $.each(arrFile, function(key, item){
                html += '<li class="flex flex-wrap items-center">';
                html += '<div class="inline-block">';
                html += '<img src="' + BASE_URL + 'frontend/template/img/icon-file.png" alt="" class="w-[30px]">';
                html += '</div>';
                html += '<div class="font-bold w-[calc(100%-50px)] px-2.5">' + item.name + '</div>';
                html += '<div class="w-5 text-[16px] text-right removeFile" data-remove="'+ key +'">';
                html += '<i class="fa-solid fa-trash-can"></i>';
                html += '</div>';
                html += '</li>';
            });
        }else{
            $('.btn-upload-file').removeClass('hasFile');
        }

        $('.btn-upload-file').text(textBtn);
        $('.listFileUpload').append(html);
    }
