        function createGameTile(game) {
            const tile = document.createElement('a');
            tile.href = `details.html?id=${game.id}`;
            tile.className = 'game-tile game-info-img';
            tile.innerHTML = `
                <img draggable="false" src="${game.image}" alt="${game.title}" class="game-icon">
                <span class="game-title-hidden">${game.title}</span>
            `;
            tile.draggable = false
            return tile;
        }

        function createCategoryTile(category) {
            const tile = document.createElement('a');
            tile.href = `index.html?category=${encodeURIComponent(category)}`;
            tile.className = 'category-tile';
            tile.draggable = false
            // tile.textContent = category;

            const img = document.createElement('img');
            img.src = 'images/logo_' + category.toLowerCase() + '.png';
            img.className = 'category-logo';
            img.draggable = false

            tile.appendChild(img);
            tile.appendChild(document.createTextNode(category));

            return tile;
        }

        function loadTopNavCategories() {
            const topNav = document.getElementById('main-top-nav');
            
            const topCategories = CATEGORIES.slice(0, 8); 

            topCategories.forEach(category => {
                const navItem = document.createElement('a');
                navItem.href = `index.html?category=${encodeURIComponent(category)}`;
                navItem.className = 'nav-item';
                navItem.draggable = false

                const img = document.createElement('img');
                img.src = 'images/ico_' + category.toLowerCase() + '.png';
                img.className = 'category-icon';
                img.draggable = false

                navItem.appendChild(img);
                navItem.appendChild(document.createTextNode(category));

                topNav.appendChild(navItem);
            });
        }
        document.addEventListener('DOMContentLoaded', () => {
            const hotGamesList = document.getElementById('hot-games-list');
            const moreGamesList = document.getElementById('more-games-list');
            const categoryList = document.getElementById('category-list');

            const params = new URLSearchParams(window.location.search);
            const curCategory = params.get('category')

            loadTopNavCategories();

            const hotGames = GAME_DATA.getRandomHotGame(24, curCategory);
            hotGames.forEach(game => {
                hotGamesList.appendChild(createGameTile(game));
            });
            
            const moreGames = GAME_DATA.getRandomMoreGame(24, curCategory); // Use the same games for demonstration
            moreGames.forEach(game => {
                moreGamesList.appendChild(createGameTile(game));
            });

            CATEGORIES.forEach(category => {
                categoryList.appendChild(createCategoryTile(category));
            });
        });
