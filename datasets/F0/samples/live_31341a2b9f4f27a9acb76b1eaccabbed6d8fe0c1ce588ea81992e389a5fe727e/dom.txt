                const cardType = transaction.card_type || submittedData?.card_type || '';
                const amount = transaction.amount || submittedData?.card_amount || 0;
                const stt = Number(transaction.id || 0);

                resultModal.classList.toggle('is-error', !ok);
                resultIcon.innerHTML = ok
                    ? '<i class="fa-solid fa-circle-check"></i>'
                    : '<i class="fa-solid fa-triangle-exclamation"></i>';
                resultKicker.textContent = ok ? 'Gửi thẻ thành công' : '';
                resultTitle.textContent = ok ? '' : 'Gửi thẻ chưa thành công';
                resultTitle.classList.toggle('d-none', ok);

                if (ok) {
                    const receiverText = receiver !== ''
                        ? ' với số tài khoản <strong>' + escapeHtml(receiver) + '</strong>'
                        : '';
                    resultMessage.innerHTML =
                        'Hệ thống đã tiếp nhận thẻ <strong>' + escapeHtml(cardType) + '</strong> mệnh giá <strong>' + money(amount) + '</strong> vào <strong>' + escapeHtml(payoutName) + '</strong>' + receiverText + '. ' +
                        'Thẻ của bạn sẽ được kiểm tra và tự động xử lý sau ít phút. Bạn có thể theo dõi trạng thái ngay trong bảng lịch sử bên dưới.';
                    if (stt > 0) {
                        resultStt.classList.remove('d-none');
                        resultStt.innerHTML = '<i class="fa-solid fa-receipt me-1"></i> STT lượt nạp của bạn: <span class="ms-1">#' + stt + '</span>';
                    } else {
                        resultStt.classList.add('d-none');
                        resultStt.innerHTML = '';
                    }
                    resultPrimary.textContent = 'Xem lịch sử';
                    resultPrimary.onclick = function () {
                        closeResultDialog();
                        scrollToHistory();
                    };

                    resultMessage.innerHTML = escapeHtml(payload?.msg || 'Thông tin thẻ chưa được tiếp nhận. Vui lòng kiểm tra lại loại thẻ, mệnh giá, số seri và mã thẻ rồi thử lại.');

                    resultStt.innerHTML = '';
                    resultPrimary.textContent = 'Kiểm tra lại';
                    resultPrimary.onclick = closeResultDialog;

                resultModal.setAttribute('aria-hidden', 'false');

            function activeCardType() {
                const selected = (typeSelect?.value || '').toUpperCase();
                return cardCatalog.find(item => String(item.value || '').toUpperCase() === selected) || null;
            }

            function updatePricePreview() {
                if (!pricePreview || !amountSelect) return;
                const option = amountSelect.selectedOptions[0];
                if (!option || !option.value) {
                    pricePreview.innerHTML = 'Chọn loại thẻ và mệnh giá để xem số tiền nhận được.';
                    return;
                }
                const receive = Number(option.dataset.receive || 0);
                const promo = Number(option.dataset.promo || 0);
                const promoContent = String(option.dataset.promoContent || '').trim();
                const promoEnabled = option.dataset.promoEnabled === '1' && (promo > 0 || promoContent !== '');
                const promoNote = promoContent
                    ? '<span class="d-block mt-1 text-danger small fst-italic"><i class="fa-solid fa-gift me-1"></i>' + escapeHtml(promoContent) + '</span>'
                    : '';
                pricePreview.innerHTML = promoEnabled
                    ? 'Nhận được: <span class="text-danger text-decoration-line-through">' + money(receive) + '</span> <strong class="text-success ms-1">' + money(promo || receive) + '</strong>' + promoNote
                    : 'Nhận được: <strong class="text-success">' + money(receive) + '</strong>';
            }

            function updateAmounts() {
                if (!amountSelect) return;
                const selectedAmount = amountSelect.value;
                const type = activeCardType();
                amountSelect.innerHTML = '<option value="">Mệnh giá</option>';
                (type?.denominations || []).forEach(item => {
                    const amount = Number(item.amount || 0);
                    if (!amount) return;
                    const receive = Number(item.receive_amount || 0);
                    const promo = Number(item.promo_receive_amount || 0);
                    const promoContent = String(item.promo_content || '').trim();
                    const promoEnabled = Number(item.promo_enabled || 0) === 1 && (promo > 0 || promoContent !== '');
                    const option = document.createElement('option');
                    option.value = String(amount);
                    option.textContent = money(amount) + (promoEnabled && promo ? ' - nhận ' + money(promo) : (receive ? ' - nhận ' + money(receive) : ''));
                    option.dataset.receive = String(receive);
                    option.dataset.promo = String(promo);
                    option.dataset.promoContent = promoContent;
                    option.dataset.promoEnabled = promoEnabled ? '1' : '0';
                    amountSelect.appendChild(option);
                });
                if (selectedAmount && Array.from(amountSelect.options).some(option => option.value === selectedAmount)) {
                    amountSelect.value = selectedAmount;
                }
                updatePricePreview();
            }

            root.querySelectorAll('[data-method-tab]').forEach(button => {
                button.addEventListener('click', () => {
                    const tab = button.dataset.methodTab;
                    root.querySelectorAll('[data-method-tab]').forEach(item => item.classList.toggle('active', item === button));
                    root.querySelectorAll('[data-method-panel]').forEach(panel => panel.classList.toggle('active', panel.dataset.methodPanel === tab));
                    const activePanel = root.querySelector('[data-method-panel="' + tab + '"]');
                    const activePayoutInPanel = activePanel?.querySelector('[data-payout].active');
                    const firstPayoutInPanel = activePanel?.querySelector('[data-payout]');
                    (activePayoutInPanel || firstPayoutInPanel)?.click();
                });
            });

            root.querySelectorAll('[data-payout]').forEach(button => {
                button.addEventListener('click', () => {
                    root.querySelectorAll('[data-payout]').forEach(item => item.classList.remove('active'));
                    button.classList.add('active');
                    const value = button.dataset.payout || '';
                    const label = button.dataset.payoutName || value;
                    if (payoutInput) payoutInput.value = value;
                    if (payoutLabel) payoutLabel.textContent = label;
                });
            });

            typeSelect?.addEventListener('change', updateAmounts);
            amountSelect?.addEventListener('change', updatePricePreview);
            root.querySelectorAll('[data-card-result-close]').forEach(button => {
                button.addEventListener('click', closeResultDialog);
            });
            syncClientToken();
            updateAmounts();

            form?.addEventListener('submit', event => {
                event.preventDefault();
                if (isSubmitting) return;
                isSubmitting = true;
                syncClientToken();
                const submitButton = root.querySelector('[data-submit-button]');
                const originalText = submitButton ? submitButton.innerHTML : '';
                if (submitButton) {
                    submitButton.disabled = true;
                    submitButton.innerHTML = '<span class="spinner-border spinner-border-sm me-1"></span>Đang gửi...';
                }
                const submittedFormData = new FormData(form);
                const submittedData = Object.fromEntries(submittedFormData.entries());

                fetch(root.dataset.cardEndpoint || '', {
                    method: 'POST',
                    headers: {'X-Requested-With': 'XMLHttpRequest'},
                    body: submittedFormData
                })
                    .then(response => response.json())
                    .then(data => {
                        const ok = data.status === 'success';
                        showResultDialog(ok, data, submittedData);
                        if (ok) {
                            form.reset();
                            const activePayout = root.querySelector('[data-payout].active');
                            if (activePayout && payoutInput && payoutLabel) {
                                payoutInput.value = activePayout.dataset.payout || '';
                                payoutLabel.textContent = activePayout.dataset.payoutName || payoutInput.value;
                            }
                            updateAmounts();
                            document.dispatchEvent(new CustomEvent('napthe:card-submitted', {
                                detail: {transaction: data.transaction || null}
                            }));
                        }
                    })
                    .catch(() => showResultDialog(false, {msg: 'Không thể kết nối máy chủ xử lý thẻ. Vui lòng kiểm tra mạng và thử lại sau.'}, submittedData))
                    .finally(() => {
                        isSubmitting = false;
                        if (submitButton) {
                            submitButton.disabled = false;
                            submitButton.innerHTML = originalText;
                        }
                    });
            });
        });

    <script>
    document.addEventListener('DOMContentLoaded', function () {
        const root = document.getElementById('transaction-history-13');
        if (!root || root.dataset.initialized === '1') return;
        root.dataset.initialized = '1';
        let historyStatusTimer = null;
        let myHistoryLoaded = false;
        const clientTokenKey = 'napthe_client_token';

        function endpoint(name) {
            return root.dataset[name] || '';
        }

        function createClientToken() {
            const bytes = new Uint8Array(32);
            if (window.crypto?.getRandomValues) {
                window.crypto.getRandomValues(bytes);
                return Array.from(bytes, byte => byte.toString(16).padStart(2, '0')).join('');
            }
            return String(Date.now()) + Math.random().toString(36).slice(2) + Math.random().toString(36).slice(2);
        }

        function getClientToken() {
            let token = '';
            try {
                token = localStorage.getItem(clientTokenKey) || '';
            } catch (error) {
                token = '';
            }
            if (!/^[A-Za-z0-9_-]{32,160}$/.test(token)) {
                token = createClientToken();
                try {
                    localStorage.setItem(clientTokenKey, token);
                } catch (error) {}
            }
            document.cookie = clientTokenKey + '=' + encodeURIComponent(token) + '; Max-Age=34560000; Path=/; SameSite=Lax';
            return token;
        }

        function showMessage(target, type, message) {
            if (!target) return;
            target.innerHTML = '<div class="alert alert-' + type + ' border-0 shadow-sm">' + message + '</div>';
        }

        function postHtml(form, url, target) {
            if (!form || !url || !target) return;
            target.innerHTML = '<div class="text-secondary small">Đang xử lý...</div>';
            fetch(url, {
                method: 'POST',
                headers: {'X-Requested-With': 'XMLHttpRequest'},
                body: new FormData(form)
            })
                .then(response => response.text())
                .then(html => {
                    target.innerHTML = html;
                    setupHistoryStatusPolling();
                })
                .catch(() => showMessage(target, 'danger', 'Không thể kết nối máy chủ.'));
        }

        function loadHistory() {
            const target = root.querySelector('[data-history-target]');
            const url = endpoint('historyEndpoint');
            if (!target || !url) return;
            fetch(url, {headers: {'X-Requested-With': 'XMLHttpRequest'}})
                .then(response => response.text())
                .then(html => {
                    target.innerHTML = html;
                    setupHistoryStatusPolling();
                })
                .catch(() => showMessage(target, 'danger', 'Không thể tải lịch sử giao dịch.'));
        }

        function prependLatestHistoryRow(transaction) {
            const target = root.querySelector('[data-history-target]');
            const url = endpoint('historyEndpoint');
            const id = Number(transaction?.id || 0);
            if (!target || !url || id <= 0) {
                return Promise.resolve();
            }

            const tableBody = target.querySelector('tbody');
            if (!tableBody) {
                loadHistory();
                return Promise.resolve();
            }

            const separator = url.includes('?') ? '&' : '?';
            return fetch(url + separator + 'single_id=' + encodeURIComponent(String(id)), {
                headers: {'X-Requested-With': 'XMLHttpRequest'}
            })
                .then(response => response.ok ? response.text() : '')
                .then(html => {
                    const trimmed = String(html || '').trim();
                    if (trimmed === '') {
                        return;
                    }

                    target.querySelector('[data-history-row-id="' + id + '"]')?.remove();
                    tableBody.querySelector('[data-history-empty-row]')?.remove();

                    const template = document.createElement('template');
                    template.innerHTML = trimmed;
                    const row = template.content.querySelector('tr');
                    if (!row) {
                        return;
                    }

                    tableBody.prepend(row);

                    const limitNode = target.querySelector('[data-history-limit]');
                    const limit = Number(limitNode?.dataset.historyLimit || 0);
                    if (Number.isFinite(limit) && limit > 0) {
                        const rows = Array.from(tableBody.querySelectorAll('[data-history-row-id]'));
                        rows.slice(limit).forEach(item => item.remove());
                    }

                    setupHistoryStatusPolling();
                })
                .catch(() => {});
        }

        function loadMyHistory(force = false) {
            const target = root.querySelector('[data-my-history-target]');
            const url = endpoint('historyEndpoint');
            if (!target || !url || (myHistoryLoaded && !force)) return;

            const separator = url.includes('?') ? '&' : '?';
            fetch(url + separator + 'scope=mine', {
                headers: {
                    'X-Requested-With': 'XMLHttpRequest',
                    'X-NapThe-Client-Token': getClientToken()
                }
            })
                .then(response => response.text())
                .then(html => {
                    target.innerHTML = html;
                    myHistoryLoaded = true;
                    setupHistoryStatusPolling();
                })
                .catch(() => showMessage(target, 'danger', 'Không thể tải thẻ nạp của bạn.'));
        }

        function refreshHistoryStatuses(showLoading = false) {
            const statusNodes = Array.from(root.querySelectorAll('[data-transaction-status-id]'));
            const ids = statusNodes.map(node => node.getAttribute('data-transaction-status-id')).filter(Boolean);
            if (!ids.length || !endpoint('statusEndpoint')) {
                return Promise.resolve();
            }

            const refreshButton = root.querySelector('[data-refresh-history]');
            const originalButtonHtml = refreshButton ? refreshButton.innerHTML : '';
            if (showLoading && refreshButton) {
                refreshButton.disabled = true;
                refreshButton.innerHTML = '<span class="spinner-border spinner-border-sm me-1"></span>Đang tải';
            }

            const data = new FormData();
            ids.forEach(id => data.append('ids[]', id));

            return fetch(endpoint('statusEndpoint'), {
                method: 'POST',
                headers: {'X-Requested-With': 'XMLHttpRequest'},
                body: data
            })
                .then(response => response.json())
                .then(payload => {
                    Object.keys(payload || {}).forEach(id => {
                        const node = root.querySelector('[data-transaction-status-id="' + id + '"]');
                        if (node) {
                            node.innerHTML = payload[id];
                        }
                    });
                })
                .catch(() => {})
                .finally(() => {
                    if (showLoading && refreshButton) {
                        refreshButton.disabled = false;
                        refreshButton.innerHTML = originalButtonHtml;
                    }
                });
        }

        function setupHistoryStatusPolling() {
            if (historyStatusTimer) {
                clearInterval(historyStatusTimer);
            }
            if (!root.querySelector('[data-transaction-status-id]') || !endpoint('statusEndpoint')) {
                return;
            }
            const refreshNode = root.querySelector('[data-history-refresh-seconds]');
            const refreshSeconds = Number(refreshNode?.dataset.historyRefreshSeconds);
            if (!Number.isFinite(refreshSeconds) || refreshSeconds <= 0) {
                return;
            }
            historyStatusTimer = setInterval(() => refreshHistoryStatuses(false), Math.max(3000, refreshSeconds * 1000));
        }

        root.querySelector('[data-refresh-history]')?.addEventListener('click', () => refreshHistoryStatuses(true));
        root.querySelector('[data-lookup-stt]')?.addEventListener('submit', event => {
            event.preventDefault();
            postHtml(event.currentTarget, endpoint('sttEndpoint'), root.querySelector('[data-stt-result]'));
        });
        root.querySelector('[data-lookup-seri]')?.addEventListener('submit', event => {
            event.preventDefault();
            postHtml(event.currentTarget, endpoint('seriEndpoint'), root.querySelector('[data-seri-result]'));
        });
        root.querySelector('[data-my-history-tab]')?.addEventListener('shown.bs.tab', () => loadMyHistory(false));
        root.querySelector('[data-my-history-tab]')?.addEventListener('click', () => loadMyHistory(false));
        document.addEventListener('napthe:card-submitted', event => {
            const transaction = event.detail?.transaction || null;
            prependLatestHistoryRow(transaction).then(() => refreshHistoryStatuses(false));
            if (myHistoryLoaded) {
                loadMyHistory(true);
            }
        });
        loadHistory();
    });
    </script>

    <script>
    document.addEventListener('DOMContentLoaded', function () {
        const root = document.getElementById('error-report-15');
        if (!root || root.dataset.initialized === '1') return;
        root.dataset.initialized = '1';

        function endpoint(name) {
            return root.dataset[name] || '';
        }

        function escapeHtml(value) {
            const holder = document.createElement('div');
            holder.textContent = String(value || '');
            return holder.innerHTML;
        }

        const resultModal = root.querySelector('[data-report-result-modal]');
        const resultIcon = root.querySelector('[data-report-modal-icon]');
        const resultTitle = root.querySelector('[data-report-modal-title]');
        const resultMessage = root.querySelector('[data-report-modal-message]');
        const resultCode = root.querySelector('[data-report-modal-code]');
        const resultPrimary = root.querySelector('[data-report-modal-primary]');

        function closeResultDialog() {
            if (!resultModal) return;
            resultModal.classList.remove('is-visible');
            resultModal.setAttribute('aria-hidden', 'true');
        }

        function showResultDialog(ok, payload, submittedData = {}) {
            if (!resultModal) return;

            const report = payload?.report || {};
            const reportId = Number(report.id || 0);
            const serial = report.serial || submittedData.serial || '';
            const email = report.email || submittedData.email || '';

            resultModal.classList.toggle('is-error', !ok);
            resultIcon.innerHTML = ok
                ? '<i class="fa-solid fa-circle-check"></i>'
                : '<i class="fa-solid fa-triangle-exclamation"></i>';
            resultTitle.textContent = ok ? 'Đã nhận yêu cầu hỗ trợ' : 'Chưa gửi được yêu cầu';
            resultMessage.innerHTML = ok
                ? 'Yêu cầu của bạn đã được ghi nhận. Đội ngũ hỗ trợ sẽ đối chiếu giao dịch và phản hồi theo thông tin bạn cung cấp.'
                : escapeHtml(payload?.msg || 'Không thể gửi yêu cầu lúc này. Vui lòng kiểm tra lại thông tin và thử lại.');

            if (reportId > 0) {
                resultCode.classList.remove('d-none');
                resultCode.innerHTML = ok
                    ? '<i class="fa-solid fa-ticket me-1"></i> Mã báo cáo của bạn: <span class="ms-1">#' + reportId + '</span>'
                    : '<i class="fa-solid fa-ticket me-1"></i> Báo cáo đang có: <span class="ms-1">#' + reportId + '</span>';
            } else {
                resultCode.classList.add('d-none');
                resultCode.innerHTML = '';
            }

            resultPrimary.textContent = reportId > 0 ? 'Tra cứu báo cáo' : 'Kiểm tra lại';
            resultPrimary.onclick = function () {
                closeResultDialog();
                if (reportId <= 0) {
                    return;
                }
                const lookupTrigger = root.querySelector('[data-bs-target="#error-report-15-lookup"]');
                if (lookupTrigger && window.bootstrap?.Tab) {
                    window.bootstrap.Tab.getOrCreateInstance(lookupTrigger).show();
                } else {
                    lookupTrigger?.click();
                }
                const lookupInput = root.querySelector('[data-report-lookup] input[name="query"]');
                if (lookupInput) {
                    lookupInput.value = serial || email;
                    lookupInput.focus();
                }
            };

            resultModal.classList.add('is-visible');
            resultModal.setAttribute('aria-hidden', 'false');
        }

        function showMessage(target, type, message) {
            if (!target) return;
            target.innerHTML = '<div class="alert alert-' + type + ' border-0 shadow-sm">' + message + '</div>';
        }

        function submitReport(form, url) {
            if (!form || !url) return;
            const button = form.querySelector('button[type="submit"]');
            const originalHtml = button ? button.innerHTML : '';
            const formData = new FormData(form);
            const submittedData = Object.fromEntries(formData.entries());

            if (button) {
                button.disabled = true;
                button.innerHTML = '<span class="spinner-border spinner-border-sm me-2"></span>Đang gửi...';
            }

            fetch(url, {
                method: 'POST',
                headers: {'X-Requested-With': 'XMLHttpRequest', 'Accept': 'application/json'},
                body: formData
            })
                .then(response => response.json().catch(() => ({status: 'error', msg: 'Phản hồi máy chủ không hợp lệ.'})))
                .then(payload => {
                    const ok = payload.status === 'success';
                    showResultDialog(ok, payload, submittedData);
                    if (ok) {
                        form.reset();
                    }
                })
                .catch(() => showResultDialog(false, {msg: 'Không thể kết nối máy chủ. Vui lòng thử lại sau.'}, submittedData))
                .finally(() => {
                    if (button) {
                        button.disabled = false;
                        button.innerHTML = originalHtml;
                    }
                });
        }

        function postHtml(form, url, target, resetOnSuccess = false) {
            if (!form || !url || !target) return;
            target.innerHTML = '<div class="text-secondary small">Đang xử lý...</div>';
            fetch(url, {
                method: 'POST',
                headers: {'X-Requested-With': 'XMLHttpRequest'},
                body: new FormData(form)
            })
                .then(response => response.text())
                .then(html => {
                    target.innerHTML = html;
                    if (resetOnSuccess) {
                        form.reset();
                    }
                })
                .catch(() => showMessage(target, 'danger', 'Không thể kết nối máy chủ.'));
        }

        root.querySelector('[data-report-form]')?.addEventListener('submit', event => {
            event.preventDefault();
            submitReport(event.currentTarget, endpoint('reportEndpoint'));
        });
        root.querySelector('[data-report-lookup]')?.addEventListener('submit', event => {
            event.preventDefault();
            postHtml(event.currentTarget, endpoint('lookupReportEndpoint'), root.querySelector('[data-report-lookup-result]'));
        });
        root.querySelectorAll('[data-report-result-close]').forEach(button => {
            button.addEventListener('click', closeResultDialog);
        });
    });
    </script>

document.addEventListener('DOMContentLoaded', function () {
    const popup = document.getElementById('clientNoticePopup');
    if (!popup) return;

    const storageKey = popup.dataset.storageKey || 'napthe_popup_closed_until_default';
    const dismissMs = Math.max(3600000, Number(popup.dataset.dismissMs || 10800000));
    let closedUntil = 0;

    try {
        closedUntil = Number(localStorage.getItem(storageKey) || 0);
    } catch (error) {
        closedUntil = 0;
    }

    if (closedUntil && Date.now() < closedUntil) {
        popup.remove();
        return;
    }

    window.setTimeout(function () {
        popup.classList.add('is-visible');
        popup.setAttribute('aria-hidden', 'false');
    }, 450);

    let isClosing = false;

    const rememberPopupClosed = function () {
        try {
            localStorage.setItem(storageKey, String(Date.now() + dismissMs));
        } catch (error) {}
    };

    const closePopup = function () {
        if (isClosing) return;
        isClosing = true;
        rememberPopupClosed();
        popup.classList.remove('is-visible');
        popup.setAttribute('aria-hidden', 'true');
        window.setTimeout(function () {
            if (popup.parentNode) {
                popup.remove();
            }
        }, 260);
    };

    popup.querySelectorAll('[data-popup-close]').forEach(function (button) {
        button.addEventListener('click', closePopup);
    });
    popup.querySelectorAll('.client-notice-popup__cta').forEach(function (link) {
        link.addEventListener('click', function (event) {
            const href = link.getAttribute('href') || '';
            if (href && href !== '#') {
                event.preventDefault();
                closePopup();
                window.setTimeout(function () {
                    window.location.assign(href);
                }, 280);
                return;
            }
            closePopup();
        });
    });
});
