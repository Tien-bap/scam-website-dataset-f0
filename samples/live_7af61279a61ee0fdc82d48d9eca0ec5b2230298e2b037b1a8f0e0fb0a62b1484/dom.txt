    <script>

        document.addEventListener('DOMContentLoaded', function () {

            const stickyBar =
                document.getElementById('lyt-sticky-buy-bar');

            const addCartButton =
                document.getElementById('lyt-sticky-add-cart');

            const buyNowButton =
                document.getElementById('lyt-sticky-buy-now');

            const stickyPrice =
                document.getElementById('lyt-sticky-price-value');

            /*
             * Form mua hàng WooCommerce gốc
             */
            const cartForm =
                document.querySelector('form.cart');

            if (!stickyBar || !cartForm) {
                return;
            }


            /*
             * Nút Add to Cart gốc
             */
            const originalButton =
                cartForm.querySelector(
                    '.single_add_to_cart_button'
                );

            if (!originalButton) {
                return;
            }


            /* =================================================
             * HIỆN BAR KHI ĐÃ CUỘN QUA NÚT MUA GỐC
             * ================================================= */

            const observer =
                new IntersectionObserver(function(entries) {

                    const entry = entries[0];

                    /*
                     * Chỉ hiện nếu nút mua đã đi lên phía trên màn hình
                     *
                     * Không hiện khi nút mua còn nằm phía dưới
                     */
                    const passedButton =
                        !entry.isIntersecting &&
                        entry.boundingClientRect.bottom < 0;


                    if (passedButton) {

    stickyBar.classList.add('lyt-show');
    document.body.classList.add('lyt-sticky-active');

} else {

    stickyBar.classList.remove('lyt-show');
    document.body.classList.remove('lyt-sticky-active');
}

                }, {
                    threshold: 0
                });


            observer.observe(originalButton);


            /* =================================================
             * KIỂM TRA VARIABLE PRODUCT
             * ================================================= */

            function buttonIsReady() {

                if (
                    originalButton.disabled ||
                    originalButton.classList.contains('disabled') ||
                    originalButton.classList.contains(
                        'wc-variation-selection-needed'
                    )
                ) {

                    /*
                     * Nếu chưa chọn biến thể
                     * đưa người dùng về form gốc
                     */
                    cartForm.scrollIntoView({
                        behavior: 'smooth',
                        block: 'center'
                    });

                    return false;
                }

                return true;
            }


            /* =================================================
             * THÊM GIỎ
             * ================================================= */

            addCartButton.addEventListener(
                'click',
                function() {

                    if (!buttonIsReady()) {
                        return;
                    }


                    /*
                     * Xóa flag Mua ngay nếu có
                     */
                    const oldFlag =
                        cartForm.querySelector(
                            'input[name="lyt_buy_now"]'
                        );

                    if (oldFlag) {
                        oldFlag.remove();
                    }


                    /*
                     * Click nút WooCommerce gốc
                     */
                    originalButton.click();

                }
            );


            /* =================================================
             * MUA NGAY
             * ================================================= */

            buyNowButton.addEventListener(
                'click',
                function() {

                    if (!buttonIsReady()) {
                        return;
                    }


                    /*
                     * Thêm flag Mua ngay vào form WooCommerce
                     */
                    let buyNowFlag =
                        cartForm.querySelector(
                            'input[name="lyt_buy_now"]'
                        );


                    if (!buyNowFlag) {

                        buyNowFlag =
                            document.createElement('input');

                        buyNowFlag.type = 'hidden';
                        buyNowFlag.name = 'lyt_buy_now';

                        cartForm.appendChild(buyNowFlag);
                    }


                    buyNowFlag.value = '1';


                    /*
                     * Submit bằng chính nút WooCommerce
                     */
                    originalButton.click();

                }
            );


            /* =================================================
             * VARIABLE PRODUCT:
             * CẬP NHẬT GIÁ STICKY
             * ================================================= */

            if (
                window.jQuery &&
                stickyPrice
            ) {

                jQuery(document).on(
                    'found_variation',
                    'form.variations_form',
                    function(event, variation) {

                        if (
                            variation &&
                            variation.price_html
                        ) {

                            stickyPrice.innerHTML =
                                variation.price_html;
                        }

                    }
                );

            }

        });

    </script>

(function () {
  if (window.tpltCheckoutStyleV1) return;
  window.tpltCheckoutStyleV1 = true;

  function tidyCheckout() {
    if (!document.body.classList.contains('woocommerce-checkout')) return;

    var form = document.querySelector('form.checkout');
    if (!form) return;

    var button = form.querySelector('#place_order');

    if (button) {
      /* Chỉ đổi chữ của hai phương thức đang có */
      var selected = form.querySelector(
        'input[name="payment_method"]:checked'
      );

      if (selected && /^(bacs|cod)$/.test(selected.value)) {
        if (button.tagName === 'INPUT') {
          button.value = 'Đặt hàng';
        } else {
          button.textContent = 'Đặt hàng';
        }
        button.setAttribute('data-value', 'Đặt hàng');
      }
    }

    var terms = form.querySelector('.form-row.terms');
    var label = terms && terms.querySelector('label[for="terms"]');

    if (label && !label.dataset.tpltTranslated) {
      var link = label.querySelector('a');

      if (link) {
        link.textContent = 'điều khoản và điều kiện';
        label.replaceChildren(
          document.createTextNode('Tôi đã đọc và đồng ý với '),
          link,
          document.createTextNode('.')
        );
        label.dataset.tpltTranslated = '1';
      }
    }

    /* Di chuyển ô đồng ý lên trước nút đặt hàng */
    if (terms && button && terms.parentNode === button.parentNode) {
      button.parentNode.insertBefore(terms, button);
    }
  }

  function init() {
    tidyCheckout();

    /* WooCommerce có thể dựng mới phần thanh toán */
    if (window.jQuery) {
      window.jQuery(document.body).on(
        'updated_checkout.tpltStyle payment_method_selected.tpltStyle',
        tidyCheckout
      );
    }
  }

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', init);
  } else {
    init();
  }
})();
</script>

<script>
(function () {
  function initCartInfo() {
    if (!document.body.classList.contains('woocommerce-cart')) return;

    var mobile = window.matchMedia('(max-width: 767px)');
    var items = [];

    document.querySelectorAll('.woocommerce .info-cart').forEach(function (box, index) {
      var heading = box.querySelector('.widgettitle');
      var content = box.querySelector('.textwidget');

      if (!heading || !content || box.dataset.tpltReady) return;
      box.dataset.tpltReady = '1';

      var originalTitle = heading.textContent.trim();
      var button = document.createElement('button');
      button.type = 'button';
      button.className = 'tplt-info-toggle';

      var shortTitle = originalTitle;
      if (/thanh toán/i.test(originalTitle)) shortTitle = 'Hình thức thanh toán';
      if (/giao hàng/i.test(originalTitle)) shortTitle = 'Phí giao hàng';

      if (!content.id) content.id = 'tplt-cart-info-' + index;
      button.textContent = shortTitle;
      button.setAttribute('aria-controls', content.id);

      button.addEventListener('click', function () {
        var expanded = button.getAttribute('aria-expanded') === 'true';
        button.setAttribute('aria-expanded', String(!expanded));
        content.hidden = expanded;
      });

      items.push({
        heading: heading,
        content: content,
        button: button,
        originalTitle: originalTitle
      });
    });

    function updateLayout() {
      items.forEach(function (item) {
        if (mobile.matches) {
          item.heading.replaceChildren(item.button);
          item.button.setAttribute('aria-expanded', 'false');
          item.content.hidden = true;
        } else {
          item.heading.textContent = item.originalTitle;
          item.content.hidden = false;
        }
      });
    }

    updateLayout();
    mobile.addEventListener('change', updateLayout);
  }

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', initCartInfo);
  } else {
    initCartInfo();
  }
})();
</script>
<script>
(function () {
  /* Tránh chạy trùng */
  if (window.tpltAutoUpdateCart) return;
  window.tpltAutoUpdateCart = true;

  var timer;

  function scheduleUpdate(event) {
    if (!document.body.classList.contains('woocommerce-cart')) return;

    var input = event.target;

    if (!input.matches('table.cart input.qty')) return;

    clearTimeout(timer);

    var form = input.closest('form');
    if (!form) return;

    timer = setTimeout(function () {
      if (!form.isConnected) return;

      /* Chỉ cập nhật khi số lượng hợp lệ */
      var quantities = form.querySelectorAll('input.qty');

      for (var i = 0; i < quantities.length; i++) {
        if (
          quantities[i].value.trim() === '' ||
          !quantities[i].checkValidity()
        ) {
          return;
        }
      }

      var button = form.querySelector('[name="update_cart"]');
      if (!button) return;

      /* Kích hoạt đúng nút cập nhật của WooCommerce */
      button.disabled = false;
      button.removeAttribute('aria-disabled');
      button.click();
    }, 700);
  }

  document.addEventListener('input', scheduleUpdate);
  document.addEventListener('change', scheduleUpdate);

  /* Hủy lần tự cập nhật đang chờ nếu khách đã bấm nút */
  document.addEventListener('click', function (event) {
    if (
      event.target.closest &&
      event.target.closest('body.woocommerce-cart [name="update_cart"]')
    ) {
      clearTimeout(timer);
    }
  });
})();
</script>
