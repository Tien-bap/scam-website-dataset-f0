        const mobileMenu = document.getElementById('mobileMenu');

        function setMobileMenu(open) {
            mobileMenu.classList.toggle('show', open);
            mobileMenuIcon.classList.toggle('bi-list', !open);
            mobileMenuIcon.classList.toggle('bi-x-lg', open);
            mobileMenuToggle.setAttribute('aria-expanded', open ? 'true' : 'false');
            mobileMenuToggle.setAttribute('aria-label', open ? 'Đóng menu' : 'Mở menu');
        }

        mobileMenuToggle.addEventListener('click', function() {
            setMobileMenu(!mobileMenu.classList.contains('show'));
        });
        
        // Close mobile menu when clicking outside
        document.addEventListener('click', function(event) {
            const toggle = event.target.closest('.mobile-toggle');
            if (!mobileMenu.contains(event.target) && !toggle && mobileMenu.classList.contains('show')) {
                setMobileMenu(false);
            }
        });

        mobileMenu.querySelectorAll('a').forEach(function(link) {
            link.addEventListener('click', function() {
                setMobileMenu(false);
            });
        });

        document.addEventListener('keydown', function(event) {
            if (event.key === 'Escape' && mobileMenu.classList.contains('show')) {
                setMobileMenu(false);
                mobileMenuToggle.focus();
            }
        });

document.addEventListener('DOMContentLoaded', function() {
    const overlay = document.getElementById('popupNotificationOverlay');
    const modal = overlay?.querySelector('.popup-notification-modal');
    if (!overlay || !modal) return;

    const popupDisabledUntil = localStorage.getItem('popupDisabledUntil');
    if (popupDisabledUntil && Number(popupDisabledUntil) > Date.now()) {
        return;
    }

    overlay.hidden = false;
    document.body.classList.add('has-popup-notification');
    document.getElementById('popupCloseBtn').addEventListener('click', closePopup);
    document.getElementById('popupBtnClose').addEventListener('click', closePopup);

    document.getElementById('popupBtnDisable').addEventListener('click', function() {
        localStorage.setItem('popupDisabledUntil', String(Date.now() + (60 * 60 * 1000)));
        closePopup();
    });

    overlay.addEventListener('click', function(e) {
        if (e.target === overlay) {
            closePopup();
        }
    });

    document.addEventListener('keydown', function(e) {
        if (e.key === 'Escape' && !overlay.hidden) closePopup();
    });

    function closePopup() {
        overlay.classList.add('is-closing');
        document.body.classList.remove('has-popup-notification');
        window.setTimeout(() => {
            overlay.hidden = true;
            overlay.classList.remove('is-closing');
        }, 180);
    }
});
