      function openModal() {
         var modal = document.getElementById('searchModal');
         modal.style.display = 'flex';
         document.body.style.overflow = 'hidden';
         // Trigger reflow para animação
         void modal.offsetWidth;
         modal.classList.add('show');
         setTimeout(function() {
            var input = modal.querySelector('.search-overlay-bar input');
            if (input) input.focus();
         }, 150);
      }

      function closeModal() {
         var modal = document.getElementById('searchModal');
         modal.classList.remove('show');
         document.body.style.overflow = '';
         setTimeout(function() {
            modal.style.display = '';
         }, 300);
      }

      // Fechar ao clicar no fundo

         var modal = document.getElementById('searchModal');
         if (e.target === modal) closeModal();

      // Fechar com ESC
      document.addEventListener('keydown', function(e) {
         if (e.key === 'Escape') {
            var modal = document.getElementById('searchModal');
            if (modal.classList.contains('show')) closeModal();
         }
      });

      // Fix submenus mobile
      document.addEventListener('DOMContentLoaded', function() {
         var sidenav = document.querySelector('.u-sidenav');
         if (!sidenav) return;

         // 1) Proteger .u-nav-popup contra transitionend de elementos filhos
         sidenav.querySelectorAll('.u-nav-popup').forEach(function(popup) {
            ['transitionend', 'webkitTransitionEnd', 'oTransitionEnd'].forEach(function(evt) {
               popup.addEventListener(evt, function(e) {
                  if (e.target !== popup) {
                     e.stopImmediatePropagation();
                     e.stopPropagation();
                  }
               }, true);
            });
         });

         // 2) Observar classe 'open' nos popups para toggle de .submenu-open no li pai
         var observer = new MutationObserver(function(mutations) {
            mutations.forEach(function(m) {
               if (m.attributeName !== 'class') return;
               var popup = m.target;
               if (!popup.classList.contains('u-nav-popup')) return;
               var li = popup.parentElement;
               if (!li) return;
               if (popup.classList.contains('open')) {
                  li.classList.add('submenu-open');
               } else {
                  li.classList.remove('submenu-open');
               }
            });
         });
         sidenav.querySelectorAll('.u-nav-popup').forEach(function(popup) {
            observer.observe(popup, {
               attributes: true,
               attributeFilter: ['class']
            });
         });

         // 3) Prevenir fechamento do sidenav ao interagir com campo de busca mobile
         var mobileSearchWrapper = document.querySelector('.mobile-search-wrapper');
         if (mobileSearchWrapper) {
            var mobileInput = mobileSearchWrapper.querySelector('input[type="text"]');

            // Bloqueia propagação de eventos de interação (prev. de clicks acidentais)
            ['click', 'touchstart', 'touchend', 'touchmove', 'mousedown', 'mouseup', 'pointerdown', 'pointerup'].forEach(function(evt) {
               mobileSearchWrapper.addEventListener(evt, function(e) { e.stopPropagation(); }, true);
            });

            if (mobileInput) {
               mobileInput.addEventListener('focus', function(e) { e.stopPropagation(); }, true);
               mobileInput.addEventListener('blur',  function(e) { e.stopPropagation(); }, true);
               mobileInput.addEventListener('touchend', function(e) {
                  e.stopPropagation();
                  e.preventDefault();
                  this.focus();
               }, { passive: false, capture: true });
            }
         }

         // 4) O menu mobile deve fechar SOMENTE via botão X ou clique no overlay.
         //    O nicepage fecha o menu em vários cenários indesejados:
         //    - resize da janela (teclado virtual no mobile)
         //    - ESC
         //    - clique em link de submenu com href="#"
         //    Solução: sobrescrever ResponsiveMenu.prototype.close e só deixar passar
         //    quando a ação vier explicitamente do X (.u-menu-close) ou do overlay (.u-menu-overlay).
         if (window.ResponsiveMenu) {
            var _closeAllowed = false;

            // Listener em fase de captura (roda ANTES dos handlers jQuery):
            // marca _closeAllowed = true apenas quando a origem é X ou overlay
            document.addEventListener('click', function(e) {
               _closeAllowed = !!(e.target && e.target.closest &&
                  (e.target.closest('.u-menu-close') || e.target.closest('.u-menu-overlay')));
            }, true);

            var _origClose = window.ResponsiveMenu.prototype.close;
            window.ResponsiveMenu.prototype.close = function(menu, instantly) {
               if (_closeAllowed) {
                  _closeAllowed = false;
                  _origClose.call(this, menu, instantly);
               }
               // else: silencia sem fechar (resize, ESC, href="#", etc.)
            };
         }

         // 5) Indicador de navegação ativo (marca o item correspondente à rota atual)
         (function() {
            var currentPath = window.location.pathname;

            function setActive(link) {
               link.classList.add('nav-active');
               // Marca o item pai (nível 1) quando um sub-item está ativo
               var popup = link.closest('.u-nav-popup');
               if (popup) {
                  var parentItem = popup.closest('.u-nav-item');
                  if (parentItem) {
                     var parentLink = parentItem.querySelector(':scope > .u-nav-link');
                     if (parentLink) parentLink.classList.add('nav-active-parent');
                  }
               }
            }

            var selectors = [
               '.u-nav-container .u-nav-link',
               '.u-nav-container-collapse .u-nav-link'
            ];

            document.querySelectorAll(selectors.join(',')).forEach(function(link) {
               var href = link.getAttribute('href');
               if (!href || href === '#' || href.indexOf('javascript') === 0) return;
               // Ignora links que contêm apenas imagem (ex: logo no sidenav)
               if (link.querySelector('img') && !link.textContent.trim()) return;
               try {
                  var url = new URL(href, window.location.href);
                  // Ignora links externos (outro domínio)
                  if (url.origin !== window.location.origin) return;
                  var lp = url.pathname.replace(/\/$/, '') || '/';
                  var cp = currentPath.replace(/\/$/, '') || '/';
                  // Correspondência exata ou prefixo de segmento
                  if (lp.length > 1 && (cp === lp || cp.startsWith(lp + '/'))) {
                     setActive(link);
                  }
               } catch (err) {}
            });
         })();
