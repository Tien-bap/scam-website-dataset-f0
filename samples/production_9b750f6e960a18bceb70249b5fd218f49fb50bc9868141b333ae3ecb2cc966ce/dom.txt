    (function () {
        'use strict';

        var pageSlug = "6-2";
        var currentUrl = new URL(window.location.href);
        var currentKeyword = currentUrl.searchParams.get('keyword');
        var keyword = currentKeyword && currentKeyword.trim() ? currentKeyword.trim() : pageSlug;

        function updateInternalLink(anchor) {
            var rawHref = anchor.getAttribute('href');

            if ( !rawHref || rawHref.charAt(0) === '#' || /^(?:mailto:|tel:|sms:|javascript:)/i.test(rawHref) ) {
                return;
            }

            var destination;

            try {
                destination = new URL(rawHref, window.location.href);
            } catch (error) {
                return;
            }

            if (
                !/^https?:$/.test(destination.protocol) ||
                destination.origin !== window.location.origin ||
                /^\/(?:wp-admin(?:\/|$)|wp-login\.php(?:\/|$))/i.test(destination.pathname)
            ) {
                return;
            }

            var linkKeyword = destination.searchParams.get('keyword');

            if (linkKeyword === null || linkKeyword.trim() === '') {
                destination.searchParams.set('keyword', keyword);
                anchor.setAttribute('href', destination.href);
            }
        }

        function handleLinkEvent(event) {
            var anchor = event.target.closest ? event.target.closest('a[href]') : null;

            if (anchor) {
                updateInternalLink(anchor);
            }
        }

        document.addEventListener('pointerdown', handleLinkEvent, true);
        document.addEventListener('click', handleLinkEvent, true);
        document.addEventListener('auxclick', handleLinkEvent, true);
        document.addEventListener('contextmenu', handleLinkEvent, true);
        document.addEventListener('keydown', function (event) {
            if (event.key === 'Enter') {
                handleLinkEvent(event);
            }
        }, true);
    })();
