(function(){
  function formatNum(n){
    if(n>=1000000) return (n/1000000).toFixed(1).replace('.',',')+'tr+';
    if(n>=1000) return Math.round(n/1000)+'k+';
    return n+'%';
  }
  var nums = document.querySelectorAll('.stat-number[data-target]');
  var started = false;
  function startCount(){
    if(started) return;
    started = true;
    nums.forEach(function(el){
      var target = parseInt(el.getAttribute('data-target'));
      var start = 0;
      var duration = 1800;
      var step = target / (duration / 16);
      var cur = 0;
      var timer = setInterval(function(){
        cur += step;
        if(cur >= target){ cur = target; clearInterval(timer); }
        el.textContent = formatNum(Math.round(cur));
      }, 16);
    });
  }
  var section = document.querySelector('.stats-section');
  if(section){
    var obs = new IntersectionObserver(function(entries){
      if(entries[0].isIntersecting){ startCount(); obs.disconnect(); }
    },{threshold:0.3});
    obs.observe(section);
  }
})();
